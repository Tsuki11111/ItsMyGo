# 一次买够 · 本地启动清单
#
# 说明：本文件是「按顺序复制执行」的命令清单，不是一键脚本。
# 后端、前端、客服智能体需要各开一个终端（都会一直占着终端）。
# 前置：Docker Desktop 已启动。

# ─────────────────────────────────────────────
# 0. 中间件（mysql / redis / minio）
# ─────────────────────────────────────────────
# 这三个容器是多个项目共用的，不归本项目的 Compose 管理。
# Docker Desktop 重启后 redis / minio 不会自动拉起，mysql 被手动 stop 过也不会。

docker start mysql redis minio

# 确认三个都在跑
docker ps --filter "name=^(mysql|redis|minio)$" --format "table {{.Names}}\t{{.Status}}"

# 可选：等 MySQL 真正可连（容器起来 ≠ 数据库就绪）
until docker exec mysql sh -c 'mysqladmin ping -uroot -p"$MYSQL_ROOT_PASSWORD" --silent' 2>/dev/null; do
  echo "等待 MySQL 就绪..."
  sleep 2
done
echo "MySQL 已就绪"


# ─────────────────────────────────────────────
# 0.5 RocketMQ（可选，本项目自有容器）
# ─────────────────────────────────────────────
# 超时未支付订单自动取消：应用连 proxy 的 8081（gRPC），namesrv 的 9876 只给 broker/proxy 用。
# 不启动 MQ 时在 .env 设 ORDER_TIMEOUT_MQ_ENABLED=false，数据库扫描兜底仍会关单。
cd /c/Users/28470/IdeaProjects/ItsMyGo
docker compose -f docker-compose.rocketmq.yml up -d
docker ps --filter "name=^its-my-go-rocketmq" --format "table {{.Names}}\t{{.Status}}\t{{.Ports}}"

# 等 broker 注册成功（有 broker-a 输出即可）：
docker exec its-my-go-rocketmq-broker sh mqadmin clusterList -n its-my-go-rocketmq-namesrv:9876

# 等 topic-init 建完 topic（状态 Exited (0) 再启动后端；gRPC 客户端启动时要求 topic 已存在）：
docker ps -a --filter "name=^its-my-go-rocketmq-topic-init" --format "table {{.Names}}\t{{.Status}}"

# topic 由 compose 里的一次性 topic-init 服务自动创建（消息类型 DELAY）；手工补建（可选）：
# docker exec its-my-go-rocketmq-broker sh mqadmin updateTopic \
#   -n its-my-go-rocketmq-namesrv:9876 -c DefaultCluster -t its-my-go-order-timeout \
#   -a +message.type=DELAY

# 停止（保留数据卷）：
# docker compose -f docker-compose.rocketmq.yml down

# ─────────────────────────────────────────────
# 2.5 客服智能体「买小购」（终端 3，占用 8003）
# ─────────────────────────────────────────────
# 让消息页的「买小购」可对话。依赖 Mongo / Milvus / Neo4j 等容器，
# 这些容器不在本项目 Compose 里，需要时先确认它们都在跑。
cd /d/PycharmProjects/rag_helper
.venv/Scripts/python.exe -m app.agent.api.grpc_service

# 本项目 .env 必须有（密钥取自 rag_helper/admin-key.txt 的第一行，64 位）：
# AGENT_GRPC_TARGET=127.0.0.1:8003
# AGENT_GRPC_KEY=<64 位密钥>
# AGENT_GRPC_DEADLINE_SECONDS=240
#
# 改完 .env 要重启后端；密钥绝不能提交进 Git。
#
# 订单查询（M7）：agent 会回调 Java 的 /internal/agent/**，
# 所以后端必须先在 8080 跑着；agent 侧地址默认 http://127.0.0.1:8080，
# 需要改就在 rag_helper/.env 里设 ITS_MYGO_BASE_URL。
#
# 智能下单 / 退款（M9/M10）：agent 只准备草稿，用户在聊天卡片确认后 Java 才执行；
# 一键停用：在 .env 设 AGENT_ACTIONS_ENABLED=false 并重启后端。
#
# 催发货（M8）：对待发货（已支付未发货）订单，直接对买小购说「帮我催一下订单 XXX 发货」；
# 用户在确认卡片确认后，由买小购调用发送工具给商家发站内消息，并回复用户完成 + 代商家致歉；
# 10 分钟内不重复催同一单。
#
# 前后端共用仓库根目录的 .env：后端由 spring.config.import 读取，
# 前端由 vite.config.ts 的 envDir 读取；只有 VITE_ 前缀会进浏览器。
#
# 虚拟物流（M7.5）：默认 120 分钟送达；想快速看完整轨迹，启动后端时设
# LOGISTICS_DELIVERY_MINUTES=2。订单详情页地图需要腾讯位置服务 Web 端
# （JavaScript API GL）Key：在仓库根目录 .env 配 VITE_TMAP_KEY，然后重启前端。
#
# 发货地址（M7.6）：商家在「商家后台 → 店铺资料 → 发货地址」设置，平台在
# 「管理后台 → 商户管理 → 发货地址」代设（平台自营 id=1 同一入口）。
# 点「解析坐标」需要该 Key 开通 WebServiceAPI 且当日额度未用完；
# 解析失败或额度用尽仍可保存地址，发货时回退虚拟转运中心。

# 5. 常见问题

#    → 中间件没起，执行第 0 步。这就是本次后端退出的原因。

#    netstat -ano | grep ":8080" | grep LISTENING
#    taskkill //PID <PID> //F

#    布隆过滤器只在启动时预热，重启后端即可。
